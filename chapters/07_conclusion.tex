\section{Kết luận và hướng phát triển}

\subsection{Kết luận}
\begin{itemize}[leftmargin=1.2cm]
    \item Đã trình bày thiết kế I2C Master theo RD1139: kiến trúc, thanh ghi, FSM và mô phỏng.
    \item Bổ sung minh họa ASIC flow Sky130 qua LibreLane với sign-off DRC/LVS/STA/IPVT.
\end{itemize}

\subsection{Hạn chế và hướng mở rộng}
\textbf{Hạn chế:} sign-off bằng open-source tool mang tính minh họa, không bao phủ mọi corner thương mại.

\textbf{Hướng mở rộng:} mở rộng kiểm chứng (random/SVA), đo đạc trên FPGA, và tối ưu timing/area.
